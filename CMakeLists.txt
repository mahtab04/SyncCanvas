cmake_minimum_required(VERSION 3.24)
project(SyncCanvas VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SYNCCANVAS_BUILD_CLIENT "Build the VectorGL desktop client" ON)
option(SYNCCANVAS_LINT_DEPENDENCIES "Run VectorGL's optional clang-tidy checks" OFF)
include(FetchContent)
find_package(Threads REQUIRED)
FetchContent_Declare(asio GIT_REPOSITORY https://github.com/chriskohlhoff/asio.git
    GIT_TAG 03ae834edbace31a96157b89bf50e5ee464e5ef9)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_MakeAvailable(asio json)
add_library(synccanvas_options INTERFACE)
if(MSVC)
    target_compile_options(synccanvas_options INTERFACE /W4 /permissive-)
else()
    target_compile_options(synccanvas_options INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_library(synccanvas_core src/core/Protocol.cpp src/core/Board.cpp src/core/DocumentFile.cpp)
target_include_directories(synccanvas_core PUBLIC include)
target_link_libraries(synccanvas_core PUBLIC nlohmann_json::nlohmann_json PRIVATE synccanvas_options)
add_library(synccanvas_net src/net/Connection.cpp src/net/NetworkClient.cpp)
target_include_directories(synccanvas_net SYSTEM PUBLIC "${asio_SOURCE_DIR}/asio/include")
target_compile_definitions(synccanvas_net PUBLIC ASIO_STANDALONE ASIO_NO_DEPRECATED)
target_link_libraries(synccanvas_net PUBLIC synccanvas_core Threads::Threads PRIVATE synccanvas_options)
if(WIN32)
    target_compile_definitions(synccanvas_net PUBLIC _WIN32_WINNT=0x0601 NOMINMAX)
    target_link_libraries(synccanvas_net PUBLIC ws2_32 mswsock)
endif()
add_library(synccanvas_host src/server/Server.cpp src/server/HostedServer.cpp)
target_link_libraries(synccanvas_host PUBLIC synccanvas_net PRIVATE synccanvas_options)
add_executable(synccanvas_server src/server/main.cpp)
target_link_libraries(synccanvas_server PRIVATE synccanvas_host synccanvas_options)
if(SYNCCANVAS_BUILD_CLIENT)
    # The desktop owns its windowing dependency independently of VectorGL examples.
    find_package(glfw3 3.3 CONFIG QUIET)
    if(NOT TARGET glfw)
        set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
        set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
        set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git
            GIT_TAG 7482de6071d21db77a7236155da44c172a7f6c9e)
        FetchContent_MakeAvailable(glfw)
    endif()
    set(VECTORGL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(VECTORGL_BUILD_APPS OFF CACHE BOOL "" FORCE)
    set(VECTORGL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(VECTORGL_ENABLE_HINTED_TEXT OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(vectorgl GIT_REPOSITORY https://github.com/mahtab04/vectorgl.git
        GIT_TAG dd1983cab1f3900c21b721373f8e3518388d1995)
    FetchContent_MakeAvailable(vectorgl)
    if(NOT SYNCCANVAS_LINT_DEPENDENCIES)
        set_property(TARGET vectorgl PROPERTY CXX_CLANG_TIDY "")
    endif()
    add_executable(synccanvas_client src/client/main.cpp src/client/Application.cpp src/client/Renderer.cpp)
    target_compile_definitions(synccanvas_client PRIVATE GLFW_INCLUDE_NONE)
    target_link_libraries(synccanvas_client PRIVATE synccanvas_host vectorgl glfw synccanvas_options)
    install(TARGETS synccanvas_client RUNTIME DESTINATION bin)
endif()
install(TARGETS synccanvas_server RUNTIME DESTINATION bin)
